"""El Alumno del sistema: POO I, POO II y Excepciones, todo junto."""


class Alumno:
    def __init__(self, dni, apellido, nombre):
        if not dni.isdigit() or len(dni) not in (7, 8):
            raise ValueError("el DNI tiene que ser de 7 u 8 números, sin puntos")
        self.dni = dni
        self.apellido = apellido
        self.nombre = nombre
        self._marcas = []

    def marcar(self, estado):
        if estado not in ("P", "A", "T", "J"):
            raise ValueError("la marca tiene que ser P, A, T o J")
        self._marcas.append(estado)

    def porcentaje(self):
        if not self._marcas:
            return 0
        vino = 0
        for marca in self._marcas:
            if marca in ("P", "T"):
                vino += 1
        return vino * 100 / len(self._marcas)

    def __str__(self):
        return f"{self.apellido}, {self.nombre} — {self.porcentaje():.1f}%"

        if not dni.isdigit() or len(dni) not in (7, 8):
            raise ValueError("el DNI tiene que ser de 7 u 8 números, sin puntos")
        self.dni = dni
        self.apellido = apellido
        self.nombre = nombre
        self._marcas = []

    def marcar(self, estado):
        if estado not in ("P", "A", "T", "J"):
            raise ValueError("la marca tiene que ser P, A, T o J")
        self._marcas.append(estado)

    def porcentaje(self):

            return 0
        vino = 0
        for marca in self._marcas:
            if marca in ("P", "T"):
                vino += 1
        return vino * 100 / len(self._marcas)

Parte 3 · Flask: la página se arma cuando alguien la pide
Sección titulada «Parte 3 · Flask: la página se arma cuando alguien la pide»


Ya tenés una función que devuelve la página. Lo que falta es alguien que, cuando el navegador
pide /alumnos, llame a esa función y le mande lo que devuelve. Eso es Flask.


sequenceDiagram
  participant N as Navegador
  participant F as Flask (app.py)
  participant M as Tus funciones
  N->>F: Pido /alumnos
  F->>M: pagina(curso)
  M-->>F: El HTML, como texto
  F-->>N: Te mando la página

 

La diferencia con el catálogo es cuándo se arma el HTML. El catálogo lo armabas una vez y
quedaba guardado en un archivo. Con Flask, se arma cada vez que alguien entra. Por eso, si Ana
tiene una marca nueva, el que recarga la página la ve.
